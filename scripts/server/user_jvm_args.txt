# ATTENTION : MineStrator lance Java lui-meme (java -Xms128M -Xmx<90% de l'offre> ...) et ce fichier est NON LU.
# Regler la memoire et les arguments Java dans le panel MineStrator. Valeurs conseillees : Xmx 8 Go + -XX:G1PeriodicGCInterval=300000
# pack_LBC - offre MineStrator 12 Go. Memoire reellement utilisee ~2-4 Go : 8 Go max laisse une bonne marge au systeme.
-Xms4G
-Xmx8G
-XX:+UseG1GC
-XX:+ParallelRefProcEnabled
-XX:MaxGCPauseMillis=200
-XX:+UnlockExperimentalVMOptions
-XX:+DisableExplicitGC
-XX:G1NewSizePercent=30
-XX:G1MaxNewSizePercent=40
-XX:G1HeapRegionSize=8M
-XX:G1ReservePercent=20
-XX:G1HeapWastePercent=5
-XX:G1MixedGCCountTarget=4
-XX:InitiatingHeapOccupancyPercent=15
-XX:G1MixedGCLiveThresholdPercent=90
-XX:G1RSetUpdatingPauseIntervalMillis=100
-XX:SurvivorRatio=32
-XX:+PerfDisableSharedMem
-XX:MaxTenuringThreshold=1
# rend au systeme la memoire inutilisee quand le serveur est au calme (verif toutes les 5 min)
-XX:G1PeriodicGCInterval=300000
